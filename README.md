    git clone https://github.com/SavedByLight-WIP/android_vendor_samsung_m55xq.git -b lineage-23.2 vendor/samsung/m55xq; git clone https://github.com/SavedByLight-WIP/lineage_device_samsung_m55xq -b lineage-23.2 device/samsung/m55xq; git clone https://github.com/SavedByLight-WIP/kernel_m55xq.git -b main kernel/samsung/m55xq

    . build/envsetup.sh; croot; brunch m55xq; croot; brunch m55xq
